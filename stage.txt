stage:
    backdrop Planets("gallery:Planets_Universe/Planets")
    life = 0
    shooting = False

    actor Spaceman:
        costume Spaceman_Fly_Shoot_L2("gallery:Aliens_Universe/Spaceman Fly Shoot L2")
        costume Spaceman_Fly_L2("gallery:Aliens_Universe/Spaceman Fly L2")

        when started:
            hide()
            setPosition(-280, -40)
            size = 25
            wait(2)
            show()

    actor Alien_Princess:
        costume Alien_Princess_Idle("gallery:Aliens_Universe/Alien Princess Idle")

        when started:
            show()
            setPosition(0, 100)
            opacity = 100
            rotationStyle = "leftRight"
            heading = -90
            size = 25
            say("Help me Spaceman!")
            for i in range(0, 20): 
                opacity -= 5
                wait(0.1)
            say("")
            shooting = True
            broadcast("newLevel")

    actor Enemy:
        costume Flying_Sphere_Droid_Activate_1("gallery:Aliens_Universe/Flying Sphere Droid Activate 1")
        costume Flying_Sphere_Droid_Deactivate_3("gallery:Aliens_Universe/Flying Sphere Droid Deactivate 3")
        costume Flying_Sphere_Droid_Deactivated_Idle("gallery:Aliens_Universe/Flying Sphere Droid Deactivated Idle")
        def enemyGo(level):
            setPosition(320,randrange(-100,101))
            setCostume(level)
            show()
            life = 5 * level
            while life > 0 :
                move(1.3)
                if(touching(Edge.left)):
                    hide()
                bounceOffEdge()
                wait(0.01)
            hide()
        when started:
            hide()
            rotationStyle = "leftRight"
            size = 50
            heading = -90
            showVariable(ref life)
        
        when signalReceived("newLevel"):
            enemyGo(1)
            enemyGo(2)
            enemyGo(3)
            

    actor Blue_Projectile:
        costume Blue_Projectile_3("gallery:Effects_Universe/Blue Projectile 3")

        when started:
            hide()
            size = 65
        
        when keyPressed("space"):
            if shooting:
                goTo(Spaceman)
                pointTowards(Pointer.position)
                show()
                while not touching(Edge.any) and not touching(Enemy):
                    move(15)
                    wait(0.01)
                if(touching(Enemy)):
                    hide()
                    life -= 1
                wait(0.01)
                hide()
